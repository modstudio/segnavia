הַנָּמֵל

בְּאוֹר רִאשׁוֹן עַל הַמִּזְחָה נִרְאוּ הַחֲבָלִים עוֹד לַחִים מִן הַלַּיְלָה, וְהַסִּירוֹת רָבְצוּ זוֹ אֵצֶל זוֹ.רִשּׁוּמֵי הַנָּמֵל, תר״ג, עמוד י״ב. הַשּׁוֹמֵר הָלַךְ אֶת אֹרֶךְ הָרְצִיף פַּעֲמַיִם עַד שֶׁצָּלַצֵל הַפַּעֲמוֹן.

בְּרֵאשִׁית בָּרָא אֱלֹהִים אֵת הַשָּׁמַיִם וְאֵת הָאָרֶץ, וְהָאָרֶץ הָיְתָה תֹהוּ וָבֹהוּ וְחֹשֶׁךְ עַל־פְּנֵי תְהוֹם, וְרוּחַ אֱלֹהִים מְרַחֶפֶת עַל־פְּנֵי הַמָּיִם, וְהַמַּיִם הוֹלְכִים תַּחַת הַסְּפִינָה בְּמִשְׁפָּט אֶחָד שֶׁאֵינוֹ נִפְסָק, פִּסְקָה אֲרֻכָּה שֶׁמַּתְחִילָה בְּעַמּוּד אֶחָד וְנִגְמֶרֶת בָּעַמּוּד שֶׁאַחֲרָיו, וְנוֹשֵׂאת אֶת שְׁמוֹת הַחֲבָלִים וְאֶת רֵיחַ הַזֶּפֶת וְאֶת מִשְׁקַל הַמִּפְרָשׂ הַמּוּכָן וְאֶת קוֹל הַכּוֹס בַּמִּטְבָּח וְאֶת הַמַּפָּה שֶׁאֵינָהּ נִשְׁאֶרֶת שְׁטוּחָה. בְּרֵאשִׁית בָּרָא אֱלֹהִים אֵת הַשָּׁמַיִם וְאֵת הָאָרֶץ, וְהָאָרֶץ הָיְתָה תֹהוּ וָבֹהוּ וְחֹשֶׁךְ עַל־פְּנֵי תְהוֹם, וְרוּחַ אֱלֹהִים מְרַחֶפֶת עַל־פְּנֵי הַמָּיִם, וְהַמַּיִם הוֹלְכִים תַּחַת הַסְּפִינָה בְּמִשְׁפָּט אֶחָד שֶׁאֵינוֹ נִפְסָק, פִּסְקָה אֲרֻכָּה שֶׁמַּתְחִילָה בְּעַמּוּד אֶחָד וְנִגְמֶרֶת בָּעַמּוּד שֶׁאַחֲרָיו. בְּרֵאשִׁית בָּרָא אֱלֹהִים אֵת הַשָּׁמַיִם וְאֵת הָאָרֶץ, וְהָאָרֶץ הָיְתָה תֹהוּ וָבֹהוּ וְחֹשֶׁךְ עַל־פְּנֵי תְהוֹם, וְרוּחַ אֱלֹהִים מְרַחֶפֶת עַל־פְּנֵי הַמָּיִם, וְהַמַּיִם הוֹלְכִים תַּחַת הַסְּפִינָה בְּמִשְׁפָּט אֶחָד שֶׁאֵינוֹ נִפְסָק.

בְּצָהֳרַיִם נֶהֶפְכָה הָרוּחַ שְׁתֵּי נְקֻדּוֹת, וְהַטַּבָּח הוֹצִיא לֶחֶם שֶׁקָּלַט אֶת טַעַם הַמַּחְסָן. אִישׁ לֹא דִּבֵּר עַל חֲזָרָה.

הָאַשְׁמוֹרֶת הַשְּׁנִיָּה מָצְאָה יְגִיעַת אֶתְרוֹגִים וְאִגֶּרֶת בְּלִי שׁוֹלֵחַ. שְׁנֵיהֶם הֻנְּחוּ תַּחַת שֻׁלְחַן הַמַּפּוֹת.הָאִגֶּרֶת בַּנִּסְפָּח; הָאֶתְרוֹגִים נֶאֶכְלוּ.

הָעֶרֶב הֵבִיא חוֹף נָמוּךְ שֶׁדָּמָה לַחוֹף שֶׁעָזְבוּ, עַד שֶׁפַּעֲמוֹן הַחוֹף צָלַצֵל בְּצְלִיל אַחֵר. רַב הַחֹבֵל כָּתַב שֶׁפִּסְקַת הַיָּם הוֹלִיכָה אוֹתָם, וְשֶׁהֶעָמוּד הַבָּא יִפָּתַח בְּנְחִיתָה.

קָשְׁרוּ אֶת הַחֲבָלִים וְיָרְדוּ אֶל קְרָשִׁים שֶׁאֵינָם זָזִים. נַעַר מָכַר מַיִם בְּכוֹס בַּדִּיל. נוֹוֵט הַסְּפִינָה סָגַר אֶת סֵפֶר הָרוּחוֹת הַשֵּׁנִי.

בַּבֹּקֶר קָנוּ חֶבֶל, דְּיוֹ, וּמַפָּה שֶׁטָּעֲנָה שֶׁהַנָּמֵל הַבָּא הוּא שְׁלוֹשָׁה יָמִים מִזְרָחָה. הַטַּבָּח אָמַר אַרְבָּעָה. הִפְלִיגוּ בַּיּוֹם הַשְּׁלִישִׁי.

הַמַּלָּחִים שָׁרוּ פַּעַם אַחַת, וְחָזְרוּ לַעֲבוֹדָה. שׁוֹמֵר הָרְצִיף הַחָדָשׁ הָלַךְ אֶת אָרְכּוֹ פַּעֲמַיִם, כָּרִאשׁוֹן, וְהַיּוֹמָן רָשַׁם אֶת הַדָּבָר בְּלִי פֵּרוּשׁ.
